cmake_minimum_required(VERSION 3.16)
project(astra_fm1 VERSION 0.0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(astra STATIC shared/astro.c)
target_include_directories(astra PUBLIC shared)
target_compile_options(astra PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_executable(astra_tests tests/test_astro.c)
target_link_libraries(astra_tests PRIVATE astra)
enable_testing()
add_test(NAME astronomy COMMAND astra_tests)
