cmake_minimum_required(VERSION 3.16)
project(astra_fm VERSION 0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# core/: device code (integer only, no libc) shared by the firmware, the host tests and the site's WebAssembly
set(ASTRA_CORE_SRC core/astro/astro.c core/astro/sky.c core/fm/fm.c core/chart/chart.c core/synth/patch.c core/ui/gfx.c core/app/app.c core/app/app_draw.c)
add_library(astra_core STATIC ${ASTRA_CORE_SRC})
target_include_directories(astra_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(NOT MSVC)
  target_compile_options(astra_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
# Device-rule gate: core/ must build freestanding with no floating-point registers (the FM-1 has no FPU and no libm)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  add_library(astra_core_nofloat OBJECT ${ASTRA_CORE_SRC})
  target_include_directories(astra_core_nofloat PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(astra_core_nofloat PRIVATE -ffreestanding -mgeneral-regs-only -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(test_astro tests/test_astro.c)
target_include_directories(test_astro PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(test_astro PRIVATE astra_core m)
add_test(NAME astro COMMAND test_astro)
add_executable(test_sky tests/test_sky.c)
target_include_directories(test_sky PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(test_sky PRIVATE astra_core m)
add_test(NAME sky COMMAND test_sky)
# the FM engine, with the undefined-behaviour sanitizer where there is one: an overflow in the audio path fails the test
add_executable(test_fm tests/test_fm.c core/fm/fm.c)
target_include_directories(test_fm PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_fm PRIVATE -Wall -Wextra -fsanitize=undefined -fno-sanitize-recover=undefined)
  target_link_options(test_fm PRIVATE -fsanitize=undefined)
endif()
target_link_libraries(test_fm PRIVATE m)
add_test(NAME fm COMMAND test_fm)
add_executable(test_patch tests/test_patch.c)
target_link_libraries(test_patch PRIVATE astra_core m)
add_test(NAME patch COMMAND test_patch)
# the instrument core on the host: frames to look at while developing (never shown as the instrument: that is fm1-emu)
add_executable(test_app tests/test_app.c)
target_link_libraries(test_app PRIVATE astra_core)
add_test(NAME app COMMAND test_app)
add_executable(ui_shot tests/ui_shot.c)
target_link_libraries(ui_shot PRIVATE astra_core)
# the host side of the emulator gate (tools/emu_smoke.sh)
add_executable(emu_expect tests/emu_expect.c)
target_link_libraries(emu_expect PRIVATE astra_core)
add_test(NAME emu_expect_j2000 COMMAND emu_expect 10957 43200)
set_tests_properties(emu_expect_j2000 PROPERTIES PASS_REGULAR_EXPRESSION "^c75f8c80 9ecf3a3d c157dba7 ")   # J2000.0: Sun 280.37, Moon 223.32, Mercury 271.89
# generated tables must match their generator
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME tables_fresh COMMAND ${CMAKE_COMMAND} -E env python3 ${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_astro_tables.py
           WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
