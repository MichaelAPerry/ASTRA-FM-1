cmake_minimum_required(VERSION 3.16)
project(astra_fm VERSION 0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# core/: device code (integer only, no libc) shared by the firmware, the host tests and the site's WebAssembly
add_library(astra_core STATIC core/astro/astro.c core/astro/planets.c)
target_include_directories(astra_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(NOT MSVC)
  target_compile_options(astra_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
# Device-rule gate: core/ must build freestanding with no floating-point registers (the FM-1 has no FPU and no libm)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  add_library(astra_core_nofloat OBJECT core/astro/astro.c core/astro/planets.c)
  target_compile_options(astra_core_nofloat PRIVATE -ffreestanding -mgeneral-regs-only -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(test_astro tests/test_astro.c)
target_include_directories(test_astro PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(test_astro PRIVATE astra_core m)
add_test(NAME astro COMMAND test_astro)
add_executable(test_planets tests/test_planets.c)
target_include_directories(test_planets PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(test_planets PRIVATE astra_core m)
add_test(NAME planets COMMAND test_planets)
# the host side of the emulator gate (tools/emu_smoke.sh)
add_executable(emu_expect tests/emu_expect.c)
target_link_libraries(emu_expect PRIVATE astra_core)
add_test(NAME emu_expect_j2000 COMMAND emu_expect 10957 43200)
set_tests_properties(emu_expect_j2000 PROPERTIES PASS_REGULAR_EXPRESSION "^c76067cf 9ecf3a3d 64")   # J2000.0: Sun 280.37, Moon 223.33 (Scorpio)
# generated tables must match their generator
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME tables_fresh COMMAND ${CMAKE_COMMAND} -E env python3 ${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_astro_tables.py
           WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME planet_tables_fresh COMMAND ${CMAKE_COMMAND} -E env python3 ${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_planet_tables.py
           WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
